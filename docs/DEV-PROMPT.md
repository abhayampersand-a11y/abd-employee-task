## 1. Locked stack — use exactly these

| Layer | Choice | Notes |
|---|---|---|
| Framework | Next.js 16.2.12 App Router | already installed |
| Styling | Tailwind CSS v4 | already installed |
| Database | **Neon Postgres** | serverless Postgres |
| DB admin | **pgAdmin 4** connected to Neon | for inspecting data manually |
| ORM | **Prisma** | `prisma` + `@prisma/client` |
| Client state + data | **Redux Toolkit + RTK Query** | `@reduxjs/toolkit` + `react-redux` |
| Loading UX | **Skeleton loaders** | custom, no library |
| Password hashing | `bcryptjs` | pure JS, no native build step on Windows |
| Validation | `zod` | every API input validated |

Install in one go:

```bash
pnpm add @prisma/client @prisma/adapter-pg @reduxjs/toolkit react-redux bcryptjs zod server-only
pnpm add -D prisma dotenv tsx
```

`bcryptjs` 3.x ships its own types — do **not** add `@types/bcryptjs`, it is deprecated.

### Architectural decision you must follow

RTK Query is a **client-side** data layer. That means:

- Pages that show live data become **Client Components** (`"use client"`), and fetch via RTK Query hooks.
- **App Router Route Handlers** (`app/api/**/route.ts`) are the API those hooks call.
- Server Components stay for shells, layouts, and static chrome only.
- Skeletons come from RTK Query's `isLoading`, **not** from `loading.tsx`. Add `loading.tsx` only for route-transition shells.

This is a deliberate trade-off: we give up RSC data fetching to get RTK Query's cache, invalidation, and optimistic updates. Do not mix the two approaches for the same data.

## 3. Prisma schema

Create `prisma/schema.prisma` exactly as below.

> **Prisma 7 changed three things.** Verified against the installed 7.9.1:
> 1. Generator is `prisma-client` (not `prisma-client-js`) and **requires** an `output` path.
> 2. The datasource URL lives in **`prisma.config.ts`**, not in `schema.prisma`. There is no `directUrl` field any more — point the config at `DIRECT_URL` so migrations bypass the pooler.
> 3. A **driver adapter is mandatory**; `datasourceUrl` is gone. Install `@prisma/adapter-pg` and pass `adapter: new PrismaPg({ connectionString })`.

```prisma
generator client {
  provider = "prisma-client"
  output   = "../lib/generated/prisma"
}

datasource db {
  provider = "postgresql"
}

enum Role {
  ADMIN
  EMPLOYEE
}

enum UserStatus {
  INVITED
  ACTIVE
  DISABLED
}

enum TaskStatus {
  TODO
  IN_PROGRESS
  DONE
}

enum TaskPriority {
  LOW
  MEDIUM
  HIGH
}

model Company {
  id        String   @id @default(cuid())
  name      String
  industry  String?
  size      String?
  phone     String?
  address   String?
  logoUrl   String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  users   User[]
  tasks   Task[]
  invites Invite[]
}

model User {
  id String @id @default(cuid())

  // Null between signup and company onboarding. Nothing else may be null.
  companyId String?
  company   Company? @relation(fields: [companyId], references: [id], onDelete: Cascade)

  email        String  @unique
  // Login handle generated for employees, e.g. "rahul.patel". Admins sign in by email.
  employeeId   String? @unique
  passwordHash String

  firstName String
  lastName  String

  role   Role       @default(EMPLOYEE)
  status UserStatus @default(ACTIVE)

  mustChangePassword Boolean @default(false)
  // One of: indigo | violet | emerald | amber | slate — drives the Avatar component.
  avatarTone         String  @default("indigo")

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  assignedTasks Task[]        @relation("TaskAssignee")
  createdTasks  Task[]        @relation("TaskCreator")
  comments      TaskComment[]
  sessions      Session[]
  invitesSent   Invite[]      @relation("InviteCreator")

  @@index([companyId])
  @@index([companyId, status])
}

model Session {
  id        String   @id @default(cuid())
  // SHA-256 of the raw cookie value. The raw token is never stored.
  tokenHash String   @unique
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  expiresAt DateTime
  createdAt DateTime @default(now())

  @@index([userId])
  @@index([expiresAt])
}

model Invite {
  id        String  @id @default(cuid())
  companyId String
  company   Company @relation(fields: [companyId], references: [id], onDelete: Cascade)

  email     String
  firstName String
  lastName  String

  // SHA-256 of the raw token that appears in the invite URL.
  tokenHash  String    @unique
  expiresAt  DateTime
  acceptedAt DateTime?

  createdById String
  createdBy   User   @relation("InviteCreator", fields: [createdById], references: [id])

  createdAt DateTime @default(now())

  @@index([companyId])
  @@index([companyId, email])
}

model Task {
  id        String  @id @default(cuid())
  companyId String
  company   Company @relation(fields: [companyId], references: [id], onDelete: Cascade)

  title       String
  description String?
  category    String?

  status   TaskStatus   @default(TODO)
  priority TaskPriority @default(MEDIUM)
  dueDate  DateTime?

  assigneeId String
  assignee   User   @relation("TaskAssignee", fields: [assigneeId], references: [id])

  createdById String
  createdBy   User   @relation("TaskCreator", fields: [createdById], references: [id])

  completedAt DateTime?
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt

  comments TaskComment[]

  @@index([companyId, status])
  @@index([companyId, assigneeId])
  @@index([companyId, createdById])
  @@index([companyId, dueDate])
}

model TaskComment {
  id     String @id @default(cuid())
  taskId String
  task   Task   @relation(fields: [taskId], references: [id], onDelete: Cascade)

  userId String
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)

  body      String
  createdAt DateTime @default(now())

  @@index([taskId])
}
```

Then:

```bash
npx prisma migrate dev --name init
npx prisma generate
```

Add a `prisma/seed.ts` that creates one company, one admin (`admin@acme.com` / `Admin@123`), four employees, and ~15 tasks spread across statuses and priorities — so every screen has content immediately.

## 4. Authentication

Custom session auth. No auth library — the invite and generated-credential flows need full control.

**`lib/auth.ts` must export:**

| Function | Behaviour |
|---|---|
| `hashPassword(raw)` / `verifyPassword(raw, hash)` | bcryptjs, cost 10 |
| `createSession(userId)` | 32 random bytes → raw token; store SHA-256 in `Session`; set cookie |
| `getSession()` | read cookie → hash → look up `Session` + `User`; return `null` if missing/expired |
| `requireAuth()` | `getSession()` or throw 401 |
| `requireAdmin()` | `requireAuth()` + `role === ADMIN` or throw 403 |
| `requireCompany()` | `requireAuth()` + non-null `companyId` or throw 409 (onboarding incomplete) |
| `destroySession()` | delete row + clear cookie |

**Cookie:** name `taskflow_session`, `httpOnly`, `sameSite: "lax"`, `secure` in production, `maxAge` 7 days, `path: "/"`.

**Employee ID generation** — `firstname.lastname` lowercased, non-alphanumerics stripped. On collision append `2`, `3`, … Must be unique across the whole table.

**Generated password** — 12 characters from an unambiguous alphabet (no `O`/`0`/`l`/`1`). Returned in the API response **exactly once**, never stored in plaintext, never logged.

**Invite token** — 32 random bytes hex. Raw token goes in the URL, SHA-256 goes in the DB. Single use: on accept, set `acceptedAt` and create the `User` in one transaction. Expiry 7 days.

### `proxy.ts` (project root)

Optimistic redirect only — cookie presence check, no DB lookup:

- No cookie + protected route (`/dashboard`, `/tasks`, `/teams`, `/reports`, `/settings`, `/employee/*`) → redirect `/login`
- Cookie present + `/login` or `/signup` → redirect `/dashboard`
- Matcher must exclude `/api/*`, `/_next/*`, and static files

**Real authorisation still happens in every route handler.** Proxy is a UX nicety, not a security boundary — the Next docs say so explicitly.

## 5. Permission rules — enforce on the SERVER, in every handler

| Action | Admin | Employee |
|---|---|---|
| View company details | ✅ | ✅ read-only |
| Edit company details | ✅ | ❌ |
| List employees | ✅ | ✅ (for the assignee dropdown only — name + id, no email) |
| Create / disable employee | ✅ | ❌ |
| Create / resend invite | ✅ | ❌ |
| View all company tasks | ✅ | ❌ |
| View tasks assigned to self | ✅ | ✅ |
| View tasks created by self | ✅ | ✅ |
| Create a task | ✅ | ✅ |
| Change task status | ✅ any | ✅ only where `assigneeId === me` |
| Edit / delete a task | ✅ any | ✅ only where `createdById === me` |
| Comment on a task | ✅ any in company | ✅ only if assignee or creator |

**Two non-negotiable rules:**

1. **Every single query filters by `companyId` from the session.** Never take `companyId` from the request body or a query param. One missed filter leaks another company's data.
2. Hiding a button in the UI is UX, not security. The handler must re-check.

Write a `lib/permissions.ts` with `canEditTask(user, task)`, `canChangeStatus(user, task)`, `canComment(user, task)` and use them in both the handler and the UI.

## 6. API contract — `app/api/**/route.ts`

All responses JSON. Errors: `{ error: string, fieldErrors?: Record<string,string> }` with a real HTTP status. Validate every body with zod.

### Auth

| Method | Path | Body / Query | Returns |
|---|---|---|---|
| POST | `/api/auth/signup` | firstName, lastName, email, password | user (no company yet), sets cookie |
| POST | `/api/auth/login` | identifier (email **or** employeeId), password | user + company, sets cookie |
| POST | `/api/auth/logout` | — | `{ ok: true }` |
| GET | `/api/auth/me` | — | `{ user, company }` or 401 |
| POST | `/api/auth/change-password` | currentPassword, newPassword | `{ ok: true }`, clears `mustChangePassword` |

### Company

| Method | Path | Body | Returns |
|---|---|---|---|
| POST | `/api/company` | name, industry, size, phone, address | company; links it to the signed-in admin |
| GET | `/api/company` | — | company |
| PATCH | `/api/company` | partial | company — **admin only** |

### Employees

| Method | Path | Body / Query | Returns |
|---|---|---|---|
| GET | `/api/employees` | `?status=&q=&page=` | `{ items, total, page, pageSize }` |
| POST | `/api/employees` | firstName, lastName, email? | `{ employee, employeeId, password }` — **password returned once, admin only** |
| PATCH | `/api/employees/[id]` | status | employee — admin only |

### Invites

| Method | Path | Body | Returns |
|---|---|---|---|
| POST | `/api/invites` | firstName, lastName, email | `{ invite, url }` — admin only |
| GET | `/api/invites/[token]` | — | `{ company, firstName, lastName, email }` · 410 if expired/used |
| POST | `/api/invites/[token]/accept` | password | creates user, sets cookie, returns user |

### Tasks

| Method | Path | Body / Query | Returns |
|---|---|---|---|
| GET | `/api/tasks` | `?scope=mine\|created\|all` `&status=&priority=&assigneeId=&q=&page=&pageSize=` | `{ items, total, page, pageSize }` |
| POST | `/api/tasks` | title, description?, category?, assigneeId, priority, dueDate? | task |
| GET | `/api/tasks/[id]` | — | task + comments |
| PATCH | `/api/tasks/[id]` | partial | task |
| DELETE | `/api/tasks/[id]` | — | `{ ok: true }` |
| POST | `/api/tasks/[id]/comments` | body | comment |

`scope=all` is **admin only** — return 403 for employees. Employees get `mine` (assigned to them) or `created` (assigned by them).
Setting `status: DONE` sets `completedAt`; moving off `DONE` clears it.

### Stats

| Method | Path | Returns |
|---|---|---|
| GET | `/api/stats` | Role-aware. Admin: `{ totalEmployees, totalTasks, inProgress, completed, byStatus }`. Employee: `{ myTasks, dueToday, completedThisWeek, overdue }` |

## 7. Redux Toolkit architecture

```
store/
  index.ts          configureStore, RootState, AppDispatch
  hooks.ts          useAppDispatch, useAppSelector (typed)
  api.ts            createApi base — fetchBaseQuery({ baseUrl: "/api", credentials: "include" })
  services/
    auth-api.ts     injectEndpoints
    company-api.ts
    employee-api.ts
    task-api.ts
    stats-api.ts
  slices/
    ui-slice.ts     modal open/close, active filters, mobile nav
app/providers.tsx   "use client" — <Provider store={store}>
```

- **One `createApi`** with `injectEndpoints` per feature. Do not create multiple APIs.
- `tagTypes: ["Task", "Employee", "Company", "Stats", "Me"]`
- Invalidation examples:
  - `createTask` / `updateTask` / `deleteTask` → invalidate `["Task", "Stats"]`
  - `createEmployee` / `acceptInvite` → invalidate `["Employee", "Stats"]`
  - `login` / `logout` → invalidate everything
- Use `providesTags` with per-id tags on list endpoints so a single task update doesn't refetch the whole list.
- **Optimistic update required** for task status change (`onQueryStarted` + `updateQueryData` + rollback on error). Dragging a card must feel instant.
- Wire `app/providers.tsx` into `app/layout.tsx` around `{children}`.

Keep server data in RTK Query only. `ui-slice` holds UI state only — never a copy of server data.

## 8. Skeleton loaders

Create `components/ui/skeleton.tsx`:

```tsx
export function Skeleton({ className }: { className?: string }) {
  return <span className={cn("block animate-pulse rounded-md bg-slate-200", className)} />;
}
```

Then build **shape-matched** skeletons — same dimensions and spacing as the real component, so nothing shifts when data lands:

| Skeleton | Mirrors |
|---|---|
| `StatGridSkeleton` | 4 (admin) / 3 (employee) stat cards |
| `TaskTableSkeleton` | 5 rows of `TaskTable`, including the toolbar |
| `TaskBoardSkeleton` | 3 kanban columns, 3 cards each |
| `TaskCardSkeleton` | one `TaskCard` |
| `EmployeeTableSkeleton` | 5 rows of the Teams table |
| `FormSkeleton` | settings / profile cards |

Rules:

- Render on `isLoading` (first load). On `isFetching` with existing data, keep the old data and dim it — do **not** flash a skeleton on refetch.
- Skeleton block heights must match the real content, or the page jumps.
- Empty state ≠ loading state. `isLoading` → skeleton. Loaded but `items.length === 0` → the empty state.
- Error → an inline error card with a Retry button calling `refetch()`.
